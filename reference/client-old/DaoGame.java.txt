package dao.client;

import com.badlogic.gdx.ApplicationAdapter;
import com.badlogic.gdx.Gdx;
import com.badlogic.gdx.Input;
import com.badlogic.gdx.graphics.Color;
import com.badlogic.gdx.graphics.GL20;
import com.badlogic.gdx.graphics.Pixmap;
import com.badlogic.gdx.graphics.Texture;
import com.badlogic.gdx.graphics.g2d.BitmapFont;
import com.badlogic.gdx.graphics.g2d.SpriteBatch;
import com.badlogic.gdx.graphics.glutils.ShapeRenderer;
import dao.common.Realm;

/**
 * Demo OFFLINE: vòng tròn nhân vật + hào quang (glow) di chuyển bằng WASD.
 * Phím 1/2/3 đổi Cảnh Giới để thấy aura đổi màu/bán kính mà không cần asset mới.
 * Chưa có mạng: nối mạng là bước "Phiên bản Vô Danh" (docs/00-ROADMAP.md).
 */
public final class DaoGame extends ApplicationAdapter {
    private SpriteBatch batch;
    private ShapeRenderer shapes;
    private BitmapFont font;
    private Texture glow;

    private float x = 640f;
    private float y = 360f;
    private Realm realm = Realm.LUYEN_KHI;

    @Override
    public void create() {
        batch = new SpriteBatch();
        shapes = new ShapeRenderer();
        font = new BitmapFont();
        glow = createGlowTexture(128);
    }

    @Override
    public void render() {
        update(Gdx.graphics.getDeltaTime());
        draw();
    }

    private void update(float dt) {
        float dx = 0f;
        float dy = 0f;
        if (Gdx.input.isKeyPressed(Input.Keys.A)) dx -= 1f;
        if (Gdx.input.isKeyPressed(Input.Keys.D)) dx += 1f;
        if (Gdx.input.isKeyPressed(Input.Keys.S)) dy -= 1f;
        if (Gdx.input.isKeyPressed(Input.Keys.W)) dy += 1f;
        if (dx != 0f && dy != 0f) { // đi chéo không được nhanh hơn
            dx *= 0.7071f;
            dy *= 0.7071f;
        }
        x += dx * realm.moveSpeed * dt;
        y += dy * realm.moveSpeed * dt;

        if (Gdx.input.isKeyJustPressed(Input.Keys.NUM_1)) realm = Realm.LUYEN_KHI;
        if (Gdx.input.isKeyJustPressed(Input.Keys.NUM_2)) realm = Realm.TRUC_CO;
        if (Gdx.input.isKeyJustPressed(Input.Keys.NUM_3)) realm = Realm.KIM_DAN;
    }

    private void draw() {
        Gdx.gl.glClearColor(0.04f, 0.04f, 0.08f, 1f);
        Gdx.gl.glClear(GL20.GL_COLOR_BUFFER_BIT);

        Color c = new Color((realm.rgb << 8) | 0xFF); // RGB -> RGBA8888

        // Glow: vẽ texture tròn mờ nhiều lớp với blend CỘNG (SRC_ALPHA, ONE)
        batch.begin();
        batch.setBlendFunction(GL20.GL_SRC_ALPHA, GL20.GL_ONE);
        for (int i = 0; i < 3; i++) {
            float size = realm.auraRadius * 2f * (1f - i * 0.25f);
            batch.setColor(c.r, c.g, c.b, 0.35f);
            batch.draw(glow, x - size / 2f, y - size / 2f, size, size);
        }
        batch.setBlendFunction(GL20.GL_SRC_ALPHA, GL20.GL_ONE_MINUS_SRC_ALPHA);
        batch.setColor(Color.WHITE);
        // Font mặc định của libGDX KHÔNG có glyph tiếng Việt có dấu -> dùng ASCII (xem docs/03-tech-stack.md)
        font.draw(batch, "Vo Danh [" + realm.name() + "]", x - 55f, y + 28f);
        font.draw(batch, "WASD: move | 1/2/3: change realm", 12f, Gdx.graphics.getHeight() - 12f);
        batch.end();

        // Thân nhân vật
        shapes.begin(ShapeRenderer.ShapeType.Filled);
        shapes.setColor(c);
        shapes.circle(x, y, 10f);
        shapes.end();
    }

    /** Texture tròn mờ dần ra rìa, tạo bằng code (không cần file ảnh). */
    private static Texture createGlowTexture(int size) {
        Pixmap pm = new Pixmap(size, size, Pixmap.Format.RGBA8888);
        pm.setBlending(Pixmap.Blending.None);
        float r = size / 2f;
        for (int py = 0; py < size; py++) {
            for (int px = 0; px < size; px++) {
                float d = (float) Math.hypot(px + 0.5f - r, py + 0.5f - r) / r;
                float a = d >= 1f ? 0f : (1f - d) * (1f - d);
                pm.drawPixel(px, py, Color.rgba8888(1f, 1f, 1f, a));
            }
        }
        Texture t = new Texture(pm);
        t.setFilter(Texture.TextureFilter.Linear, Texture.TextureFilter.Linear);
        pm.dispose();
        return t;
    }

    @Override
    public void dispose() {
        batch.dispose();
        shapes.dispose();
        font.dispose();
        glow.dispose();
    }
}
